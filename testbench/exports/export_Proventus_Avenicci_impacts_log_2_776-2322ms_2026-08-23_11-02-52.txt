[Ingest]

; The quietest contact allowed to make any sound at all. Raise it and the small taps that fill out
; a tumble disappear; lower it and the settle at the end of a fall starts chattering.
fMinImpactSpeed = 20

; How far the solver's closing speed and our own reconstruction of it may differ before the
; contact is treated as a physics explosion and thrown away. Lower is stricter: too low and real
; hard landings go silent.
fBlowupDisagreeFrac = 0.1

; Backstop for contacts with no reconstruction to check against. Anything faster is discarded as
; impossible, so setting it low makes the hardest hits in the game the only ones that never play.
fBlowupSpeedCeiling = 1000

; The same backstop for spin. A genuine armoured landing spins a limb to 60 rad/s, so anything
; near that discards real impacts.
fBlowupAngularCeiling = 200

; Half of all contacts are one limb touching another limb of the same body. Below this they only
; thicken the cloth bed; above it they play as real impacts. Drop it and a ragdoll rattles against
; itself constantly.
fSelfContactThreshold = 250

; Every limb-on-own-limb contact is reported twice, once from each limb. Off, they all play twice
; and the whole mix doubles at the moment of landing.
bDropMirroredSelfContacts = 1

; How large a gap between contact reports means a new frame started. Only matters on replay; the
; measured split is one microsecond inside a frame against 20 ms between.
fFrameGapMs = 2

; One collision surface reported at several points becomes one contact at the strongest of them.
; Off, a flat landing on a floor plays several times over.
bCollapseManifolds = 1

; How much faster a contact must be sliding along a surface than into it before it is heard as a
; scrape rather than a thud. Lower sends more of a tumble to the scrape loop and takes the punch
; out of it.
fGrazeRatio = 1.5

; Above this closing speed a contact is always a thud, however much it is also sliding. Raise it
; and hard skids go quiet as the scrape path swallows them; lower it and genuine slides start
; thudding.
fGrazeMaxImpactSpeed = 220
