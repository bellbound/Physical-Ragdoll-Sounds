; How loud the closing flop is. It is punctuation, not an event.
fGainDb = -27

; How long after the body goes quiet the closing cue lands. Too soon and it sounds like another
; impact; too late and the fall has already been forgotten.
fDelayMs = 70


[Mix]

; The whole mod's level against the rest of the game. With vanilla's body impacts suppressed there
; is nothing to sit next to, so this calibrates against footsteps and combat - and there is no bus
; control in this engine, so it is the only lever.
fMasterGainDb = 2

; Global trim on the bright contact layer. Up for clicky and present, down for distant and soft.
fTransientTrimDb = 0

; Global trim on the flesh-and-mass layer.
fBodyTrimDb = 0

; Global trim on the pitched boom. The one control most worth touching if the mod feels either
; weightless or overbearing.
fSubTrimDb = 0

; Global trim on the wood, stone and soft colour. Up makes the floor material obvious; down makes
; every surface sound the same, which is what vanilla does.
fSurfaceTrimDb = 0

; Global trim on the small filler taps that give a burst its texture.
fGrainTrimDb = 0

; Global trim on the scrape and cloth loops.
fLoopTrimDb = 0

; Below this a sound is not worth a voice and is dropped before the cap ever sees it. Raise it to
; keep the voice budget for things that can actually be heard.
fVoiceFloorDb = -100


[Player]

; Whether your own ragdoll makes any sound at all.
bEnabled = 1

; At arm's length collapsing every layer to one point stops helping and starts sounding like the
; audio is inside your head, so your own limbs carry their own sounds.
bAttachToBones = 1

; A 30 Hz boom at zero distance through headphones is overwhelming, and in VR low frequency is
; felt as much as heard. This pulls it back for your own falls only.
fSubTrimDb = 0

; You are the one moving and the view already tells you, so the anticipation whoosh is off for
; your own falls.
bSkipAirborneWhoosh = 1

; Your own ragdoll's level, separate from everyone else's.
fMasterGainDb = -3


[Distance]

; Inside this everything plays: composites, grains, loops and bed. About ten metres.
fFullRadius = 700

; Between the two radii only the hero composites play - nobody resolves the detail at that range.
; Beyond it the actor is dropped entirely, which is what keeps a battlefield of ragdolls from ever
; becoming a performance question.
fSimplifiedRadius = 2100


[Slots]

; Pick variants out of a shuffled bag rather than at random. Random repeats itself immediately,
; and an immediate repeat is exactly what people notice.
bShuffleBag = 1

; 0 seeds from the clock, which is what the game wants. Any other value makes every run identical,
; which is what an A/B between two configs needs so it compares the configs and not two dice
; rolls.
iRngSeed = 1


[Layers]

; The contact itself - bright, fast, and the quietest layer of the stack. Mute it and an impact
; loses its edge but keeps its weight.
bImpTransient = 1

; Low-mid flesh and mass, arriving ~20 ms in. The main body of the sound at ordinary intensities,
; where the sub has not taken over yet.
bImpBody = 1

; The late pitched boom, ~65 ms in. The design's central claim is that this one layer is what
; makes a hit read as mass rather than contact - mute it and the gnarl should leave with it. If it
; does not, the balance is wrong, not the timing.
bImpSub = 1

; The hollow knock layered on when a body lands on wood. Mute the three surf_* slots together to
; hear how much of the mix is surface identity at all.
bSurfWood = 1

; The hard, short skin for stone. Vanilla plays the same dirt sample for stone, ice and carpet
; alike, so this is one of the differences the mod exists to make.
bSurfStone = 1

; The dull skin, and the fallback for every surface we cannot resolve - which today means all
; natural ground, since no dirt, grass or snow contact exists in the capture set.
bSurfSoft = 1

; The burst filler: quiet, dry, heavily pitch-scattered taps that give a burst its three to five
; grains. Mute it to hear the bursts reduced to their hero hits alone.
bLimbTap = 1

; The granular low-mid crackle that is what a bone break actually is - ten times the transient
; density of a plain thud in that band, not a snap sample.
bCrunchGran = 1

; The wet squelch, obliterate tier only. The fastest way to check the gore gate is not opening on
; ordinary knockdowns.
bGoreWet = 1

; The sliding rumble. Worth muting often: nothing in the capture set is actually a slide, so this
; is the layer most likely to be firing when it should not.
bScrapeLoop = 1

; The continuous cloth bed under everything, 30-36 dB down. Mute it to hear how much work it does
; papering over the one-shots - the references say a great deal.
bFoleyCloth = 0

; The airborne anticipation rise. Mute it to judge whether the fall still reads as a fall without
; being told one is coming.
bAirWhoosh = 0

; The dull skull thud with its granular edge. Mute it to check the head gate is firing on head
; contacts and not on everything that happens to be moving fast.
bHeadImpact = 1

; The soft final flop that closes the event. Mute it and falls should audibly trail off
; unfinished, which is the whole reason it exists.
bSettleRest = 1
